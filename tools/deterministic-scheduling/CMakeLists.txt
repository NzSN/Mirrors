cmake_minimum_required(VERSION 3.24)
project(mirrors_scheduled_counter LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(mirrorcpp CONFIG REQUIRED)
find_package(OpenSSL 3.0 REQUIRED)
if(NOT DPM_GENERATED_DIR)
  set(DPM_GENERATED_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../test/fixtures/deterministic-scheduling/generated-cpp")
endif()
add_executable(dpm_counter counter.cpp)
target_include_directories(dpm_counter PRIVATE "${DPM_GENERATED_DIR}")
target_link_libraries(dpm_counter PRIVATE mirrorcpp::mirrorcpp OpenSSL::Crypto)
target_compile_options(dpm_counter PRIVATE -Wall -Wextra -Werror)

if(DPM_NATIVE_GENERATED_DIR)
  add_executable(dpm_native native/runner.cpp)
  target_include_directories(dpm_native PRIVATE "${DPM_NATIVE_GENERATED_DIR}")
  target_link_libraries(dpm_native PRIVATE mirrorcpp::mirrorcpp OpenSSL::Crypto)
  target_compile_options(dpm_native PRIVATE -Wall -Wextra -Werror)
endif()

add_executable(dpm_explore explore.cpp)
target_include_directories(dpm_explore PRIVATE "${DPM_GENERATED_DIR}")
target_link_libraries(dpm_explore PRIVATE mirrorcpp::mirrorcpp OpenSSL::Crypto)
target_compile_options(dpm_explore PRIVATE -Wall -Wextra -Werror)

add_executable(dpm_native_adapter_test native/adapter_test.cpp)
target_link_libraries(dpm_native_adapter_test PRIVATE mirrorcpp::mirrorcpp)
target_compile_options(dpm_native_adapter_test PRIVATE -Wall -Wextra -Werror)
